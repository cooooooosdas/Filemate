async def measure_new(base, origin, fixture, output):
    rows = []
    for mode in ('recommended', 'preview'):
        for user in sorted(fixture['users'], key=lambda u: u['wrong_count']):
            for search in ([False, True] if mode == 'recommended' and user['wrong_count'] == 10000 else [False]):
                async with httpx.AsyncClient(base_url=base, timeout=30) as client:
                    login = await client.post('/api/auth/login',
                        headers={'Origin': origin, 'X-FileMate-Action': 'account'},
                        json={'email': user['email'], 'password': os.environ['FILEMATE_AUDIT_PASSWORD'], 'remember': False})
                    assert login.status_code == 200
                    async def query():
                        params = {'limit': 30, 'sort': 'recommended'} if mode == 'recommended' else {'available_minutes': 12}
                        if search:
                            params['q'] = '合成复盘10000'
                        path = '/wrongbook/v2/page' if mode == 'recommended' else '/wrongbook/v2/session-preview'
                        start = time.perf_counter()
                        response = await client.get(path, params=params)
                        assert response.status_code == 200, response.text
                        data = response.json()['data']
                        expected = 1 if search else max(0, user['wrong_count']-5)
                        assert data['total' if mode == 'recommended' else 'eligible_count'] == expected
                        assert len(data['items']) <= (30 if mode == 'recommended' else 3)
                        return (time.perf_counter()-start)*1000
                    first = await query()
                    warm = [await query() for _ in range(20)]
                    parallel = []
                    for _ in range(5):
                        parallel.extend(await asyncio.gather(*(query() for _ in range(4))))
                    def distribution(values):
                        values = sorted(values)
                        return {'samples': len(values), 'p50_ms': round(values[len(values)//2],2),
                                'p95_ms': round(values[math.ceil(len(values)*.95)-1],2)}
                    row = {'count': user['wrong_count'], 'mode': mode, 'search': search,
                        'process_first_ms': round(first,2), 'warm': distribution(warm), 'concurrent_4': distribution(parallel)}
                    row['warm_meets_500ms_p95'] = row['warm']['p95_ms'] <= 500
                    row['concurrent_4_meets_500ms_p95'] = row['concurrent_4']['p95_ms'] <= 500
                    rows.append(row)
    report = {'synthetic': True, 'target_p95_ms': 500, 'rows': rows,
        'first_request_definition': 'one fresh API process, then first request per workspace/mode; OS cache not flushed'}
    output.write_text(json.dumps(report,ensure_ascii=False,indent=2),encoding='utf-8')
    return report
